from pathlib import Path
import subprocess,shutil,signal
base=Path('/private/tmp/picfetch-maintainability-27-mac-state')
base.mkdir(exist_ok=False)
# The packaged binary keeps its production ID. Temporarily isolate only its
# preferences/session directories and restore them after its process exits.
paths=[Path('/Users/ronin/Library/Preferences/fyne/io.github.frathe.picfetch'),Path('/Users/ronin/Library/Caches/fyne/io.github.frathe.picfetch')]
existing=subprocess.run(['pgrep','-ix','picfetch'],stdout=subprocess.PIPE,stderr=subprocess.PIPE)
if existing.returncode not in (1,): raise RuntimeError('Cannot establish that the production application is stopped')
moved=[];child=None
try:
 for i,p in enumerate(paths):
  backup=base/str(i)
  if p.exists():shutil.move(str(p),str(backup));moved.append((p,backup))
 with (base/'process.log').open('w') as log:
  child=subprocess.Popen(['/private/tmp/picfetch-maintainability-27-package/bin/PicFetch.app/Contents/MacOS/picfetch','/private/tmp/picfetch-comparison-smoke/01-alpha.png'],stdout=log,stderr=subprocess.STDOUT)
  def stop(signum,frame):
   if child.poll() is None: child.terminate()
  signal.signal(signal.SIGTERM,stop);signal.signal(signal.SIGINT,stop)
  print('Packaged macOS smoke PID',child.pid,flush=True)
  code=child.wait();print('Packaged macOS smoke exit',code,flush=True)
finally:
 for i,p in enumerate(paths):
  if p.exists():shutil.move(str(p),str(base/(str(i)+'-smoke-generated')))
 for p,backup in moved:shutil.move(str(backup),str(p))
 print('Original preferences and session restored.',flush=True)
