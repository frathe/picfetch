$ErrorActionPreference = 'Stop'
$desktop = [Environment]::GetFolderPath('Desktop')
$source = Split-Path -Parent $PSScriptRoot
$result = Join-Path $desktop ('PicFetch-package-results-' + (Get-Date -Format yyyyMMdd-HHmmss))
New-Item -ItemType Directory -Path $result | Out-Null
Start-Transcript -Path (Join-Path $result 'transcript.txt')
try {
    Add-Type -TypeDefinition @'
using System;
using System.Runtime.InteropServices;
public static class PicFetchSmokeWindow {
    [DllImport("user32.dll")] public static extern uint GetDpiForWindow(IntPtr window);
}
'@
    Write-Host 'Four package windows will open in sequence. Close each window when its visual check is finished.'
    Write-Host 'Each window opens two generated test images. Right switches from Alpha to Beta.'
    Write-Host 'The application keeps its production identity; each process receives separate test preferences and cache folders.'
    $outcomes = @()
    foreach ($kind in @('windows', 'microsoft-store')) {
        foreach ($arch in @('amd64', 'arm64')) {
            $label = "$kind-$arch"
            $original = Join-Path $source "bin\picfetch-$label.exe"
            $hash = (Get-FileHash -LiteralPath $original -Algorithm SHA256).Hash
            $exe = Join-Path $desktop ("picfetch-package-$label-" + $hash.Substring(0,12) + '.exe')
            if (Test-Path -LiteralPath $exe) {
                if ((Get-FileHash -LiteralPath $exe -Algorithm SHA256).Hash -ne $hash) { throw "Different existing binary: $exe" }
            } else { Copy-Item -LiteralPath $original -Destination $exe }
            $work = Join-Path $result $label
            New-Item -ItemType Directory -Path $work | Out-Null
            $profile = Join-Path $work 'profile'
            $local = Join-Path $profile 'AppData\Local'
            $roaming = Join-Path $profile 'AppData\Roaming'
            New-Item -ItemType Directory -Path $local, $roaming | Out-Null
            Copy-Item -LiteralPath (Join-Path $PSScriptRoot '01-alpha.png'), (Join-Path $PSScriptRoot '02-beta.png') -Destination $work
            $start = New-Object System.Diagnostics.ProcessStartInfo
            $start.FileName = $exe
            $start.WorkingDirectory = $desktop
            $start.UseShellExecute = $false
            $start.RedirectStandardOutput = $true
            $start.RedirectStandardError = $true
            $start.EnvironmentVariables['USERPROFILE'] = $profile
            $start.EnvironmentVariables['LOCALAPPDATA'] = $local
            $start.EnvironmentVariables['APPDATA'] = $roaming
            $start.Arguments = '"' + (Join-Path $work '01-alpha.png') + '" "' + (Join-Path $work '02-beta.png') + '"'
            Write-Host "Opening $label; SHA256 $hash"
            $process = New-Object System.Diagnostics.Process
            $process.StartInfo = $start
            if (-not $process.Start()) { throw "Could not launch $label" }
            $stdout = $process.StandardOutput.ReadToEndAsync()
            $stderr = $process.StandardError.ReadToEndAsync()
            $deadline = (Get-Date).AddSeconds(30)
            do {
                $process.Refresh()
                if ($process.HasExited -or $process.MainWindowHandle -ne [IntPtr]::Zero) { break }
                Start-Sleep -Milliseconds 100
            } while ((Get-Date) -lt $deadline)
            $dpi = 0
            if (-not $process.HasExited -and $process.MainWindowHandle -ne [IntPtr]::Zero) {
                $dpi = [PicFetchSmokeWindow]::GetDpiForWindow($process.MainWindowHandle)
                Write-Host "Window ready: $($process.MainWindowTitle); DPI $dpi"
            }
            $process.WaitForExit()
            $stdout.Result | Set-Content -LiteralPath (Join-Path $work 'stdout.txt') -Encoding UTF8
            $stderr.Result | Set-Content -LiteralPath (Join-Path $work 'stderr.txt') -Encoding UTF8
            $record = [PSCustomObject]@{ Package=$label; SHA256=$hash; ProcessID=$process.Id; WindowDPI=$dpi; ExitCode=$process.ExitCode }
            $record | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $work 'process.json') -Encoding UTF8
            $outcomes += $record
            Write-Host "Closed $label; exit $($process.ExitCode). Visual results require separate observation."
            $process.Dispose()
        }
    }
    $outcomes | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $result 'processes.json') -Encoding UTF8
    'Completed package launch attempts. Check process exit codes and retained visual observations.' | Set-Content -LiteralPath (Join-Path $result 'COMPLETE.txt')
} catch {
    $_ | Out-String | Set-Content -LiteralPath (Join-Path $result 'FAILURE.txt')
    Write-Host "Package smoke stopped: $_"
} finally {
    Stop-Transcript
}
try {
    Copy-Item -LiteralPath $result -Destination (Join-Path $PSScriptRoot (Split-Path -Leaf $result)) -Recurse
    Write-Host 'Package logs copied to the share.'
} catch { Write-Host "Results remain at $result" }
