package main

import (
	"image"
	"image/color"
	"image/png"
	"os"
	"path/filepath"

	"golang.org/x/image/font"
	"golang.org/x/image/font/basicfont"
	"golang.org/x/image/math/fixed"
)

func label(img *image.RGBA, x, y int, text string) {
	small := image.NewRGBA(image.Rect(0, 0, 7*len(text), 16))
	d := font.Drawer{Dst: small, Src: image.NewUniform(color.White), Face: basicfont.Face7x13, Dot: fixed.P(0, 13)}
	d.DrawString(text)
	for py := 0; py < 16*12; py++ {
		for px := 0; px < small.Bounds().Dx()*12; px++ {
			c := small.RGBAAt(px/12, py/12)
			if c.A != 0 {
				img.SetRGBA(x+px, y+py, c)
			}
		}
	}
}
func write(dir, name string, img image.Image) {
	f, err := os.Create(filepath.Join(dir, name))
	if err != nil {
		panic(err)
	}
	encoder := png.Encoder{CompressionLevel: png.BestSpeed}
	if err := encoder.Encode(f, img); err != nil {
		panic(err)
	}
	if err := f.Close(); err != nil {
		panic(err)
	}
}
func main() {
	const w, h = 8192, 6144
	dir := "/private/tmp/picfetch-comparison-smoke"
	a := image.NewRGBA(image.Rect(0, 0, w, h))
	for y := range h {
		for x := range w {
			c := color.RGBA{45, 65, 90, 255}
			if ((x/256)+(y/256))%2 == 0 {
				c = color.RGBA{100, 120, 145, 255}
			}
			if x%256 < 4 || y%256 < 4 {
				c = color.RGBA{230, 230, 230, 255}
			}
			if x < 1024 && y < 1024 {
				c = color.RGBA{220, 45, 40, 255}
			}
			if x >= w-1024 && y < 1024 {
				c = color.RGBA{40, 180, 70, 255}
			}
			if x < 1024 && y >= h-1024 {
				c = color.RGBA{40, 80, 230, 255}
			}
			if x >= w-1024 && y >= h-1024 {
				c = color.RGBA{230, 190, 35, 255}
			}
			if x >= w/2-256 && x < w/2+256 && y >= h/2-256 && y < h/2+256 {
				if x%4 < 2 {
					c = color.RGBA{255,255,255,255}
				} else {
					c = color.RGBA{0,0,0,255}
				}
			}
			a.SetRGBA(x, y, c)
		}
	}
	label(a, 120, 120, "TOP LEFT")
	label(a, w-1000, 120, "TOP RIGHT")
	label(a, 120, h-500, "BOTTOM LEFT")
	label(a, w-1000, h-500, "BOTTOM RIGHT")
	label(a, w/2-550, h/2-650, "8192 x 6144")
	write(dir, "01-alpha.png", a)
	for y := h/2 - 512; y < h/2+512; y++ {
		for x := w/2 + 512; x < w/2+768; x++ {
			a.SetRGBA(x, y, color.RGBA{220, 40, 200, 255})
		}
	}
	write(dir, "02-beta.png", a)
	rotated := image.NewRGBA(image.Rect(0, 0, h, w))
	for y := range h {
		for x := range w {
			rotated.SetRGBA(h-1-y, x, a.RGBAAt(x, y))
		}
	}
	write(dir, "03-rotated.png", rotated)
}
