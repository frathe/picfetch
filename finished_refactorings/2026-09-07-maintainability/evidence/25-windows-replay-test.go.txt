// Appended to scripts/nativeguards/main_test.go using a temporary Go overlay.

func TestReturnedOfflineWindowsEvidence(t *testing.T) {
 dir := os.Getenv("PICFETCH_OFFLINE_RESULTS")
 if dir == "" { t.Fatal("PICFETCH_OFFLINE_RESULTS is required") }
 for _, name := range []string{"windows", "store"} {
  t.Run(name, func(t *testing.T) {
   s, err := suiteFor(name, "windows")
   if err != nil { t.Fatal(err) }
   var combined bytes.Buffer
   for _, pkg := range s.packages {
    suffix := strings.TrimPrefix(pkg, "./internal/")
    file := strings.ReplaceAll(suffix, "/", "-")
    if name == "store" { file = "store-" + file }
    eventDir := dir
    if file == "update" && os.Getenv("PICFETCH_OFFLINE_UPDATE_RESULTS") != "" { eventDir = os.Getenv("PICFETCH_OFFLINE_UPDATE_RESULTS") }
    inventory, err := os.ReadFile(eventDir + "/" + file + "-inventory.txt")
    if err != nil { t.Fatal(err) }
    full := "github.com/frathe/picfetch/internal/" + suffix
    for _, g := range s.guards {
     if g.Package == full && !slices.Contains(strings.Fields(string(inventory)), g.Test) {
      t.Fatalf("guard not build-selected: %s %s", full, g.Test)
     }
    }
    raw, err := os.ReadFile(eventDir + "/" + file + ".json")
    if err != nil { t.Fatal(err) }
    decoder := json.NewDecoder(bytes.NewReader(raw))
    passes := 0
    for {
     var e event
     if err := decoder.Decode(&e); errors.Is(err, io.EOF) { break } else if err != nil { t.Fatal(err) }
     if e.Package != full { t.Fatalf("unexpected package %q in %s", e.Package, file) }
     if e.Test == "" && e.Action == "pass" { passes++ }
    }
    if passes != 1 { t.Fatalf("package %s has %d terminal passes", file, passes) }
    combined.Write(raw)
   }
   var report bytes.Buffer
   if err := validateEvents(&combined, s.guards, &report); err != nil { t.Fatal(err) }
   t.Log(report.String())
  })
 }
}
