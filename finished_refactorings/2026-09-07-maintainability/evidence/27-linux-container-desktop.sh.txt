#!/bin/sh
set -eu
export DISPLAY=:99
Xvfb :99 -screen 0 1280x960x24 -dpi 96 -nolisten tcp > /results/xvfb.log 2>&1 &
xvfb_pid=$!
trap 'kill "$xvfb_pid" 2>/dev/null || true' EXIT INT TERM
attempt=0
until xdpyinfo -display :99 >/dev/null 2>&1; do
  attempt=$((attempt + 1))
  if [ "$attempt" -ge 100 ]; then exit 1; fi
  sleep .1
done
openbox > /results/openbox.log 2>&1 &
x11vnc -display :99 -forever -shared -nopw -listen 127.0.0.1 -rfbport 5900 -noxdamage > /results/x11vnc.log 2>&1 &
websockify --web=/usr/share/novnc 0.0.0.0:6080 127.0.0.1:5900 > /results/websockify.log 2>&1 &
{
  uname -a
  cat /etc/os-release
  dpkg --print-architecture
  xdpyinfo | sed -n '/dimensions:/p; /resolution:/p'
  glxinfo -B
  dpkg-query -W xvfb openbox x11vnc novnc libgl1-mesa-dri
  ldd /input/picfetch-linux-arm64
  sha256sum /input/picfetch-linux-arm64 /fixtures/01-alpha.png /fixtures/02-beta.png
} > /results/environment.txt 2>&1
printf 'Desktop ready\n' > /results/desktop-ready.txt
wait "$xvfb_pid"
