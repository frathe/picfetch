$ErrorActionPreference = 'Stop'
$sdk = 'C:\Program Files (x86)\Windows Kits\10'
$makeappx = Get-ChildItem -Path "$sdk\bin" -Filter MakeAppx.exe -Recurse -ErrorAction SilentlyContinue | Where-Object { $_.FullName -match '\\x64\\MakeAppx\.exe$' } | Sort-Object FullName -Descending | Select-Object -First 1
$signtool = Get-ChildItem -Path "$sdk\bin" -Filter SignTool.exe -Recurse -ErrorAction SilentlyContinue | Where-Object { $_.FullName -match '\\x64\\SignTool\.exe$' } | Sort-Object FullName -Descending | Select-Object -First 1
$appcert = Join-Path $sdk 'App Certification Kit\appcert.exe'
if (-not $makeappx -or -not $signtool -or -not (Test-Path -LiteralPath $appcert)) { throw 'Windows SDK MakeAppx/SignTool/WACK are required; no package or trust changes made.' }
$identity = 'OpenSourceDeveloperFloria.PicFetch'
if (Get-AppxPackage -Name $identity) { throw 'A PicFetch Store package is already installed; use a disposable Windows user for certification.' }
$principal = New-Object Security.Principal.WindowsPrincipal([Security.Principal.WindowsIdentity]::GetCurrent())
if (-not $principal.IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) { throw 'Run this certification script in the disposable VM from an Administrator shell.' }
$result = Join-Path ([Environment]::GetFolderPath('Desktop')) ('PicFetch-WACK-' + (Get-Date -Format yyyyMMdd-HHmmss))
New-Item -ItemType Directory -Path $result | Out-Null
Start-Transcript -Path (Join-Path $result 'transcript.txt')
$cert = $null
$trusted = $null
$trustAdded = $false
try {
    Get-Item $makeappx.FullName, $signtool.FullName, $appcert | Select-Object FullName, @{Name='Version';Expression={$_.VersionInfo.FileVersion}}
    Copy-Item -LiteralPath (Join-Path $PSScriptRoot 'stage-x64') -Destination $result -Recurse
    Copy-Item -LiteralPath (Join-Path $PSScriptRoot 'stage-arm64') -Destination $result -Recurse
    $packages = Join-Path $result 'packages'
    New-Item -ItemType Directory -Path $packages | Out-Null
    & $makeappx.FullName pack /v /h SHA256 /d (Join-Path $result 'stage-x64') /p (Join-Path $packages 'picfetch-x64.msix')
    if ($LASTEXITCODE -ne 0) { throw 'MakeAppx failed for x64.' }
    & $makeappx.FullName pack /v /h SHA256 /d (Join-Path $result 'stage-arm64') /p (Join-Path $packages 'picfetch-arm64.msix')
    if ($LASTEXITCODE -ne 0) { throw 'MakeAppx failed for ARM64.' }
    $bundle = Join-Path $result 'picfetch-microsoft-store.msixbundle'
    & $makeappx.FullName bundle /v /d $packages /p $bundle
    if ($LASTEXITCODE -ne 0) { throw 'MakeAppx bundle failed.' }
    $publisher = 'CN=00000000-0000-0000-0000-000000000000'
    $cert = New-SelfSignedCertificate -Type Custom -Subject $publisher -KeyUsage DigitalSignature -FriendlyName 'PicFetch offline packaging smoke' -CertStoreLocation 'Cert:\CurrentUser\My' -TextExtension @('2.5.29.37={text}1.3.6.1.5.5.7.3.3')
    $trusted = New-Object System.Security.Cryptography.X509Certificates.X509Store('TrustedPeople', 'LocalMachine')
    $trusted.Open('ReadWrite')
    $trusted.Add($cert)
    $trustAdded = $true
    & $signtool.FullName sign /fd SHA256 /sha1 $cert.Thumbprint $bundle
    if ($LASTEXITCODE -ne 0) { throw 'SignTool sign failed.' }
    & $signtool.FullName verify /pa /all /v $bundle
    if ($LASTEXITCODE -ne 0) { throw 'SignTool verification failed.' }
    & $appcert reset
    if ($LASTEXITCODE -ne 0) { throw 'WACK reset failed.' }
    & $appcert test -appxpackagepath $bundle -reportoutputpath (Join-Path $result 'wack-report.xml')
    if ($LASTEXITCODE -ne 0) { throw 'WACK rejected the bundle.' }
    'PASS MakeAppx, SignTool and WACK.' | Set-Content -LiteralPath (Join-Path $result 'SUCCESS.txt')
} catch {
    $_ | Out-String | Set-Content -LiteralPath (Join-Path $result 'FAILURE.txt')
    Write-Host "FAILED: $_"
} finally {
    # This identity was checked absent before testing; remove only its test installation.
    Get-AppxPackage -Name $identity | Remove-AppxPackage -ErrorAction Continue
    if ($cert) {
        if ($trusted) {
            try { if ($trustAdded) { $trusted.Remove($cert) } }
            catch { Write-Host "Test trust cleanup failed: $_" }
            finally { $trusted.Close() }
        }
        Remove-Item -LiteralPath ("Cert:\CurrentUser\My\" + $cert.Thumbprint) -ErrorAction Continue
    }
    Stop-Transcript
    Write-Host "Certification results: $result"
}
try { Copy-Item -LiteralPath $result -Destination (Join-Path $PSScriptRoot (Split-Path -Leaf $result)) -Recurse }
catch { Write-Host 'Share is read-only; certification results remain on the Desktop.' }
