func TestRecordedNativeZenityTransport(t *testing.T) {
 data, err := os.ReadFile(os.Getenv("PICFETCH_NATIVE_RESULTS"))
 if err != nil { t.Fatal(err) }
 var results []struct { Multiple bool; Paths []string; Output string; ExitCode int `json:"exit_code"` }
 if err := json.Unmarshal(data, &results); err != nil { t.Fatal(err) }
 if len(results) != 8 { t.Fatalf("got %d native cases, want 8", len(results)) }
 for _, result := range results {
  var exit error
  if result.ExitCode == 1 {
   command := exec.Command(os.Args[0], "-test.run=^TestZenityResult_DistinguishesCancelAndFailure$")
   command.Env = append(os.Environ(), "PICFETCH_ZENITY_EXIT_FIXTURE=cancel")
   exit = command.Run()
  } else if result.ExitCode != 0 { t.Fatalf("unexpected native exit %d", result.ExitCode) }
  output, err := zenityResult([]byte(result.Output), exit, result.Multiple)
  selected, err := decodePickedPaths(output, err)
  if err != nil { t.Fatal(err) }
  if len(selected) != len(result.Paths) { t.Fatalf("selection = %v, want %v", selected, result.Paths) }
  for i, uri := range selected { if uri.Path() != result.Paths[i] { t.Errorf("selected %q, want %q", uri.Path(), result.Paths[i]) } }
 }
}
