$ErrorActionPreference = 'Stop'
$desktop = [Environment]::GetFolderPath('Desktop')
$result = Join-Path $desktop ('PicFetch-graphics-details-' + (Get-Date -Format yyyyMMdd-HHmmss))
New-Item -ItemType Directory -Path $result | Out-Null
$report = [ordered]@{}
function Read-PEMachine([string]$path) {
    $stream = [IO.File]::OpenRead($path)
    $reader = New-Object IO.BinaryReader($stream)
    try {
        if ($reader.ReadUInt16() -ne 0x5a4d) { return 'not PE' }
        $stream.Position = 0x3c
        $offset = $reader.ReadInt32()
        $stream.Position = $offset
        if ($reader.ReadUInt32() -ne 0x4550) { return 'invalid PE signature' }
        $machine = $reader.ReadUInt16()
        $names = @{ 0x014c='x86'; 0x8664='amd64'; 0xaa64='arm64'; 0xa641='arm64ec'; 0xa64e='arm64x' }
        return ('0x{0:x4} ({1})' -f $machine, $names[[int]$machine])
    } finally { $reader.Dispose() }
}
try {
    $report.CapturedAt = (Get-Date -Format o)
    $report.PowerShellVersion = $PSVersionTable.PSVersion.ToString()
    $report.ProcessArchitecture = $env:PROCESSOR_ARCHITECTURE
    $report.EmulatedArchitecture = $env:PROCESSOR_ARCHITEW6432
    try { $report.Windows = @(Get-CimInstance Win32_OperatingSystem | Select-Object Caption, Version, BuildNumber, OSArchitecture) } catch { $report.WindowsError = $_.ToString() }
    try { $report.VideoControllers = @(Get-CimInstance Win32_VideoController | Select-Object Name, DriverVersion, DriverDate, VideoProcessor, CurrentHorizontalResolution, CurrentVerticalResolution) } catch { $report.VideoError = $_.ToString() }
    $report.DesktopLibraries = @(Get-ChildItem -LiteralPath $desktop -Filter '*.dll' -File | Sort-Object Name | ForEach-Object {
        [PSCustomObject]@{ Name=$_.Name; Machine=(Read-PEMachine $_.FullName); Bytes=$_.Length; FileVersion=$_.VersionInfo.FileVersion; ProductVersion=$_.VersionInfo.ProductVersion; SHA256=(Get-FileHash -LiteralPath $_.FullName -Algorithm SHA256).Hash }
    })
    $report.RendererOverrides = [ordered]@{}
    foreach ($name in @('FYNE_SCALE', 'FYNE_RENDERER', 'GALLIUM_DRIVER', 'MESA_LOADER_DRIVER_OVERRIDE', 'LIBGL_ALWAYS_SOFTWARE')) {
        $value = [Environment]::GetEnvironmentVariable($name)
        if ($null -ne $value) { $report.RendererOverrides[$name] = $value }
    }
    try { $report.GraphicsCompatibilityPackages = @(Get-AppxPackage | Where-Object { $_.Name -match 'OpenGL|OpenCL|Vulkan|DirectX' } | Select-Object Name, Version, Architecture) } catch { $report.PackageQueryError = $_.ToString() }
    try {
        Add-Type -TypeDefinition @'
using System;
using System.Collections.Generic;
using System.Runtime.InteropServices;
public static class PicFetchMonitorDetails {
    [StructLayout(LayoutKind.Sequential)] public struct Rect { public int Left, Top, Right, Bottom; }
    [StructLayout(LayoutKind.Sequential, CharSet=CharSet.Unicode)] struct MonitorInfo { public int Size; public Rect Monitor, Work; public uint Flags; [MarshalAs(UnmanagedType.ByValTStr, SizeConst=32)] public string Device; }
    public class Detail { public string Device; public int Left, Top, Right, Bottom, ScalePercent, ScaleResult; public bool Primary; }
    delegate bool MonitorCallback(IntPtr monitor, IntPtr dc, ref Rect rect, IntPtr data);
    [DllImport("user32.dll")] static extern bool EnumDisplayMonitors(IntPtr dc, IntPtr clip, MonitorCallback callback, IntPtr data);
    [DllImport("user32.dll", CharSet=CharSet.Unicode)] static extern bool GetMonitorInfo(IntPtr monitor, ref MonitorInfo info);
    [DllImport("shcore.dll")] static extern int GetScaleFactorForMonitor(IntPtr monitor, out int scale);
    public static Detail[] Read() {
        var rows = new List<Detail>();
        MonitorCallback callback = delegate(IntPtr monitor, IntPtr dc, ref Rect rect, IntPtr data) {
            var info = new MonitorInfo(); info.Size = Marshal.SizeOf(typeof(MonitorInfo));
            if (!GetMonitorInfo(monitor, ref info)) throw new InvalidOperationException("GetMonitorInfo failed");
            int scale; int result = GetScaleFactorForMonitor(monitor, out scale);
            rows.Add(new Detail { Device=info.Device, Left=info.Monitor.Left, Top=info.Monitor.Top, Right=info.Monitor.Right, Bottom=info.Monitor.Bottom, Primary=(info.Flags & 1)!=0, ScalePercent=scale, ScaleResult=result });
            return true;
        };
        if (!EnumDisplayMonitors(IntPtr.Zero, IntPtr.Zero, callback, IntPtr.Zero)) throw new InvalidOperationException("EnumDisplayMonitors failed");
        return rows.ToArray();
    }
}
'@
        $report.Monitors = @([PicFetchMonitorDetails]::Read())
    } catch { $report.MonitorError = $_.ToString() }
} catch { $report.Failure = $_.ToString() }
$report | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath (Join-Path $result 'graphics.json') -Encoding UTF8
try {
    Copy-Item -LiteralPath $result -Destination (Join-Path $PSScriptRoot (Split-Path -Leaf $result)) -Recurse
    Write-Host 'Graphics details copied to the share. No libraries or system settings were changed.'
} catch { Write-Host "Results remain at $result" }
