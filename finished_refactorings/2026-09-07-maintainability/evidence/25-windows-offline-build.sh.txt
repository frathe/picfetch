#!/bin/sh
set -eu
export GOOS=windows GOARCH=arm64 CGO_ENABLED=0
for pkg in wallpaper clipboard filepicker update ui/autoupdate distribution; do
  name=$(printf '%s' "$pkg" | tr / -)
  go test -c -o "offline-windows/picfetch-test-$name.exe" "./internal/$pkg"
done
for pkg in distribution ui/autoupdate; do
  name=$(printf '%s' "$pkg" | tr / -)
  go test -c -tags microsoftstore -o "offline-windows/picfetch-test-store-$name.exe" "./internal/$pkg"
done
go build -o offline-windows/picfetch-test2json.exe cmd/test2json
