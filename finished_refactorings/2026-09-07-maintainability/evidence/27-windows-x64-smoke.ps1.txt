param([ValidateSet('native', 'sse2')][string]$CpuCaps = 'native')
$ErrorActionPreference = 'Stop'
$desktop = [Environment]::GetFolderPath('Desktop')
$source = Split-Path -Parent $PSScriptRoot
$result = Join-Path $desktop ('PicFetch-x64-' + $CpuCaps + '-package-results-' + (Get-Date -Format yyyyMMdd-HHmmss))
New-Item -ItemType Directory -Path $result | Out-Null
Start-Transcript -Path (Join-Path $result 'transcript.txt')
try {
    Add-Type -TypeDefinition @'
using System;
using System.Runtime.InteropServices;
public static class PicFetchSmokeWindow {
    [DllImport("user32.dll")] public static extern uint GetDpiForWindow(IntPtr window);
}
'@
    Write-Host 'Two x64 package windows will open in sequence. Close each window when its visual check is finished.'
    Write-Host 'Each window opens two generated test images. Right switches from Alpha to Beta.'
    Write-Host 'The application keeps its production identity; each process receives separate test preferences and cache folders.'
    $outcomes = @()
    foreach ($kind in @('windows', 'microsoft-store')) {
        foreach ($arch in @('amd64')) {
            $label = "$kind-$arch"
            $original = Join-Path $source "bin\picfetch-$label.exe"
            $hash = (Get-FileHash -LiteralPath $original -Algorithm SHA256).Hash
            $work = Join-Path $result $label
            New-Item -ItemType Directory -Path $work | Out-Null
            $exe = Join-Path $work 'picfetch.exe'
            Copy-Item -LiteralPath $original -Destination $exe
            if ((Get-FileHash -LiteralPath $exe -Algorithm SHA256).Hash -ne $hash) { throw "Copied executable hash mismatch: $label" }
            $runtime = Join-Path $PSScriptRoot 'mesa-llvmpipe-x64-26.2.2'
            $manifest = Get-Content -LiteralPath (Join-Path $runtime 'runtime.json') -Raw | ConvertFrom-Json
            $transfer = Join-Path $runtime $manifest.TransferArchive
            $localArchive = Join-Path $work $manifest.TransferArchive
            Copy-Item -LiteralPath $transfer -Destination $localArchive
            if ((Get-FileHash -LiteralPath $localArchive -Algorithm SHA256).Hash -ne $manifest.TransferSHA256) { throw 'Transfer archive hash mismatch' }
            Expand-Archive -LiteralPath $localArchive -DestinationPath $work
            $library = Join-Path $work 'opengl32.dll'
            if ((Get-FileHash -LiteralPath $library -Algorithm SHA256).Hash -ne $manifest.SHA256) { throw 'Extracted runtime hash mismatch' }
            Copy-Item -LiteralPath (Join-Path $runtime 'runtime.json') -Destination $work
            $testProfile = Join-Path $work 'profile'
            $local = Join-Path $testProfile 'AppData\Local'
            $roaming = Join-Path $testProfile 'AppData\Roaming'
            New-Item -ItemType Directory -Path $local, $roaming | Out-Null
            Copy-Item -LiteralPath (Join-Path $PSScriptRoot '01-alpha.png'), (Join-Path $PSScriptRoot '02-beta.png') -Destination $work
            $start = New-Object System.Diagnostics.ProcessStartInfo
            $start.FileName = $exe
            $start.WorkingDirectory = $work
            $start.UseShellExecute = $false
            $start.RedirectStandardOutput = $true
            $start.RedirectStandardError = $true
            $start.EnvironmentVariables['USERPROFILE'] = $testProfile
            $start.EnvironmentVariables['LOCALAPPDATA'] = $local
            $start.EnvironmentVariables['APPDATA'] = $roaming
            if ($CpuCaps -eq 'sse2') { $start.EnvironmentVariables['GALLIUM_OVERRIDE_CPU_CAPS'] = 'sse2' }
            $start.EnvironmentVariables['GALLIUM_DUMP_CPU'] = '1'
            $start.Arguments = '"' + (Join-Path $work '01-alpha.png') + '" "' + (Join-Path $work '02-beta.png') + '"'
            Write-Host "Opening $label; SHA256 $hash"
            $process = New-Object System.Diagnostics.Process
            $process.StartInfo = $start
            if (-not $process.Start()) { throw "Could not launch $label" }
            $stdout = $process.StandardOutput.ReadToEndAsync()
            $stderr = $process.StandardError.ReadToEndAsync()
            $deadline = (Get-Date).AddSeconds(30)
            do {
                $process.Refresh()
                if ($process.HasExited -or $process.MainWindowHandle -ne [IntPtr]::Zero) { break }
                Start-Sleep -Milliseconds 100
            } while ((Get-Date) -lt $deadline)
            $dpi = 0
            if (-not $process.HasExited -and $process.MainWindowHandle -ne [IntPtr]::Zero) {
                $dpi = [PicFetchSmokeWindow]::GetDpiForWindow($process.MainWindowHandle)
                Write-Host "Window ready: $($process.MainWindowTitle); DPI $dpi"
            }
            $ready = [PSCustomObject]@{ Package=$label; ProcessID=$process.Id; WindowDPI=$dpi; SHA256=$hash; RuntimeSHA256=$manifest.SHA256; Runtime=$manifest.Runtime; CpuCaps=$CpuCaps }
            $ready | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $work 'ready.json') -Encoding UTF8
            try { $ready | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $PSScriptRoot 'x64-current.json') -Encoding UTF8 } catch { Write-Host 'Live status unavailable on share; final results will be copied.' }
            if (-not $process.HasExited) {
                try { @($process.Modules | Where-Object { $_.ModuleName -match 'opengl|gallium|EGL|GLES' } | Select-Object ModuleName, FileName) | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $work 'graphics-modules.json') -Encoding UTF8 } catch { $_ | Out-String | Set-Content -LiteralPath (Join-Path $work 'module-query-error.txt') }
            }
            $process.WaitForExit()
            $stdout.Result | Set-Content -LiteralPath (Join-Path $work 'stdout.txt') -Encoding UTF8
            $stderr.Result | Set-Content -LiteralPath (Join-Path $work 'stderr.txt') -Encoding UTF8
            $record = [PSCustomObject]@{ Package=$label; SHA256=$hash; ProcessID=$process.Id; WindowDPI=$dpi; ExitCode=$process.ExitCode; CpuCaps=$CpuCaps }
            $record | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $work 'process.json') -Encoding UTF8
            $outcomes += $record
            Write-Host "Closed $label; exit $($process.ExitCode). Visual results require separate observation."
            $process.Dispose()
        }
    }
    $outcomes | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $result 'processes.json') -Encoding UTF8
    'Completed package launch attempts. Check process exit codes and retained visual observations.' | Set-Content -LiteralPath (Join-Path $result 'COMPLETE.txt')
} catch {
    $_ | Out-String | Set-Content -LiteralPath (Join-Path $result 'FAILURE.txt') -Encoding UTF8
    Write-Host "Package smoke stopped: $_"
} finally {
    Stop-Transcript
}
try {
    $returned = Join-Path $PSScriptRoot (Split-Path -Leaf $result)
    New-Item -ItemType Directory -Path $returned | Out-Null
    foreach ($file in Get-ChildItem -LiteralPath $result -File) { Copy-Item -LiteralPath $file.FullName -Destination $returned }
    foreach ($directory in Get-ChildItem -LiteralPath $result -Directory) {
        $target = Join-Path $returned $directory.Name
        New-Item -ItemType Directory -Path $target | Out-Null
        foreach ($file in Get-ChildItem -LiteralPath $directory.FullName -File | Where-Object { $_.Extension -in @('.json', '.txt') }) {
            Copy-Item -LiteralPath $file.FullName -Destination $target
        }
    }
    Write-Host 'Package logs copied to the share.'
} catch { Write-Host "Results remain at $result" }
